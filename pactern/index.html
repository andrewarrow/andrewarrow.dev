// RESERVE QUOTA

Check every exit.

The repair handles invalid amounts and insufficient quota while preserving the pact.

↳ Four requested from three returns Err(InsufficientQuota).

enum ReserveError {
    InvalidAmount, InsufficientQuota
}

pub fn reserve(
    available: I64,
    amount: I64
) -> Result<I64, ReserveError> {
    if amount <= 0 {
        return Err(InvalidAmount)
    }
    if amount > available {
        return Err(InsufficientQuota)
    }
    return Ok(available - amount)
}
